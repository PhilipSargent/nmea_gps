# This next bit is the key that makes it work.
# 1. On the pixel9 select 2.4/5 Ghz compatibility, which enables WPA2-Personal security.
# 2. on the Mango:
# Set ieee80211w='0' (disable PMF expectation on OpenWrt's client interface) and hard-code the 
# radio channel to Channel 13 so apcli0 doesn't attempt to jump channels while associating:
# 1. Disable PMF / 802.11w requirements on the client interface
uci set wireless.sta.ieee80211w='0' # works on Guava, not on Papaya

# The uci set wireless.sta.ieee80211w='0' command fails on the second Mango because GL.iNet's OpenWrt 
# configuration does not always name the station (client) interface sta. On many firmware builds, 
# the station section is either an anonymous section or named sta0, wwan, or indexed under @wifi-iface.
#  For Papaya: define the client interface block manually.
# THIS MUST BE DONE WITH NO USB TETHERING to the Pixel9 !
# (But I never did get it to work - 19 Oct.2026 at Paros)
uci set wireless.sta=wifi-iface
uci set wireless.sta.device='mt7628'
uci set wireless.sta.mode='sta'
uci set wireless.sta.network='wwan'
uci set wireless.sta.ifname='apcli0'
uci set wireless.sta.ssid='PhilipPixel9'
uci set wireless.sta.encryption='psk2'
uci set wireless.sta.key='obafgkmrns'
uci set wireless.sta.ieee80211w='0'
uci set wireless.sta.disabled='0'
# Commit and reload wireless
uci commit wireless
wifi reload


# 2. Set the physical radio channel explicitly to 13 to match the Pixel
uci set wireless.mt7628.channel='13'

uci commit wireless
wifi reload

# Verify both settings are committed to startup storage
uci get wireless.sta.ieee80211w
uci get wireless.mt7628.channel
uci commit wireless
