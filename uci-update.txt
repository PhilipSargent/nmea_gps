# enable ch 13
 uci set wireless.mt7628.region='1'
 uci set wireless.mt7628.country='GB'

# This next bit is the key that makes it work.
# 1. On the pixel9 select 2.4/5 Ghz compatibility, which enablesd WPA2-Personal security.
# 2. on the Mango:
# Set ieee80211w='0' (disable PMF expectation on OpenWrt's client interface) and hard-code the # radio channel to Channel 13 so apcli0 doesn't attempt to jump channels while associating:
# 1. Disable PMF / 802.11w requirements on the client interface
uci set wireless.sta.ieee80211w='0'

# 2. Set the physical radio channel explicitly to 13 to match the Pixel
uci set wireless.mt7628.channel='13'

uci commit wireless
wifi reload

# Verify both settings are committed to startup storage
uci get wireless.sta.ieee80211w
uci get wireless.mt7628.channel
uci set wireless.sta.ieee80211w='0'
uci commit wireless
